Vault Operations Dashboard

These queries monitor numbered architecture folders, Notes, and each domain under Threshold.

Note Count Per Category Folder

TABLE WITHOUT ID
  Folder,
  length(rows) AS "Count"
FROM ""
FLATTEN split(file.folder, "/")[0] AS Root
FLATTEN choice(Root = "Threshold" AND length(split(file.folder, "/")) > 1, "Threshold/" + split(file.folder, "/")[1], Root) AS Folder
WHERE regextest("^[0-9]", Root) OR Root = "Notes" OR Root = "Threshold"
GROUP BY Folder
SORT Folder ASC

Notes Missing a Category Field

TABLE
  file.folder AS "Folder",
  file.mtime AS "Last Modified"
FROM ""
WHERE (regextest("^[0-9]", file.folder) OR startswith(file.folder, "Notes") OR startswith(file.folder, "Threshold/"))
  AND typeof(category) = "null"
SORT file.mtime DESC

Recently Modified Notes

TABLE
  file.mtime AS "Last Modified",
  file.folder AS "Folder"
FROM ""
WHERE (regextest("^[0-9]", file.folder) OR startswith(file.folder, "Notes") OR startswith(file.folder, "Threshold/"))
  AND file.mtime >= date(today) - dur(7 days)
SORT file.mtime DESC

Notes With No Outgoing Links

TABLE
  file.folder AS "Folder",
  file.mtime AS "Last Modified"
FROM ""
WHERE (regextest("^[0-9]", file.folder) OR startswith(file.folder, "Notes") OR startswith(file.folder, "Threshold/"))
  AND length(file.outlinks) = 0
SORT file.folder ASC

Full Vault Summary

TABLE WITHOUT ID
  Folder,
  length(rows) AS "Total Notes",
  length(filter(rows, (row) => typeof(row.category) = "null")) AS "Missing Category",
  max(map(rows, (row) => row.file.mtime)) AS "Latest Modified"
FROM ""
FLATTEN split(file.folder, "/")[0] AS Root
FLATTEN choice(Root = "Threshold" AND length(split(file.folder, "/")) > 1, "Threshold/" + split(file.folder, "/")[1], Root) AS Folder
WHERE regextest("^[0-9]", Root) OR Root = "Notes" OR Root = "Threshold"
GROUP BY Folder
SORT Folder ASC

Threshold Contract Gaps

TABLE
  category AS "Category",
  cluster AS "Cluster",
  confidence AS "Confidence",
  review AS "Review",
  file.folder AS "Folder"
FROM ""
WHERE !startswith(file.path, "Scripts/")
  AND !startswith(file.path, "Assets/")
  AND !startswith(file.path, "Engine/")
  AND !startswith(file.path, "_publish/")
  AND (typeof(category) = "null"
    OR typeof(tags) = "null"
    OR typeof(links) = "null"
    OR typeof(cluster) = "null"
    OR typeof(confidence) = "null"
    OR typeof(review) = "null")
SORT file.folder ASC, file.name ASC

Invalid Category Values

TABLE category AS "Invalid Category", file.folder AS "Folder"
FROM ""
WHERE category
  AND !contains(["Cycles", "Phases", "Flowers", "Roles", "Ecologies", "Weather"], category)
SORT category ASC, file.name ASC

Folder / Category Mismatches

TABLE category AS "Category", file.folder AS "Current Folder"
FROM "Threshold"
WHERE contains(["Cycles", "Phases", "Flowers", "Roles", "Ecologies", "Weather"], category)
  AND file.folder != "Threshold/" + category
SORT category ASC, file.name ASC

Review Lane

TABLE category, cluster, confidence, file.mtime AS "Modified"
FROM "Threshold/Review"
SORT confidence ASC, file.mtime DESC

Scope

  • Included: numbered top-level folders, Notes, and the six Threshold ontology folders.
  • Excluded: operational roots such as Assets, Engine, Scripts, _publish, and hidden folders.
  • Change the regular expression in each query if the category naming convention changes.
